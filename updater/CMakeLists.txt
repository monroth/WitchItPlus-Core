cmake_minimum_required(VERSION 3.22)
project(WitchItUpdater LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
 add_compile_options(/utf-8 /EHsc)
endif()
add_library(wip_crypto STATIC vendor/monocypher.c vendor/monocypher-ed25519.c)
add_library(wip_core STATIC core.cpp)
target_link_libraries(wip_core PUBLIC wip_crypto)
if(WIN32)
 add_library(wip_zip STATIC vendor/miniz.c vendor/miniz_zip.c vendor/miniz_tinfl.c vendor/miniz_tdef.c)
 add_executable(WitchItUpdater worker.cpp archive.cpp)
 target_link_libraries(WitchItUpdater PRIVATE wip_core wip_zip winhttp)
 # Test key/origin require an explicit separate build tree; never read from disk.
 if(WIP_TEST_PUBLIC_KEY)
  add_executable(WitchItUpdaterProbe tests/windows_probe.cpp)
  add_executable(WitchItUpdaterHttpProbe tests/http_probe.cpp)
  target_link_libraries(WitchItUpdaterHttpProbe PRIVATE winhttp ws2_32)
  target_compile_definitions(wip_core PRIVATE WIP_TEST_PUBLIC_KEY="${WIP_TEST_PUBLIC_KEY}")
  target_compile_definitions(WitchItUpdater PRIVATE WIP_TEST_PUBLIC_KEY="${WIP_TEST_PUBLIC_KEY}" WIP_TEST_ORIGIN="${WIP_TEST_ORIGIN}")
 endif()
else()
 add_library(wip_zip STATIC vendor/miniz.c vendor/miniz_zip.c vendor/miniz_tinfl.c vendor/miniz_tdef.c)
 add_executable(updater_test_driver ../../review-tests/p101_driver.cpp archive.cpp)
 target_link_libraries(updater_test_driver PRIVATE wip_core wip_zip)
 target_include_directories(updater_test_driver PRIVATE .)
 if(WIP_TEST_PUBLIC_KEY)
  target_compile_definitions(wip_core PRIVATE WIP_TEST_PUBLIC_KEY="${WIP_TEST_PUBLIC_KEY}")
 endif()
endif()
